## 4. Conversation store (verified)

`src/state/conversation.ts`: a pure reducer, so it runs unchanged inside React (`useReducer`), Vue (a `ref` + dispatch), Angular (a signal) or a vanilla store.

```ts
import type { ChatMessage, MessageStatus } from './types';

export interface ConversationState {
  byId: Record<string, ChatMessage>;
  order: string[]; // oldest → newest
  unread: number;
  isOpen: boolean;
  typing: boolean;
  hasMoreHistory: boolean;
}

export type ConversationAction =
  | { type: 'sendStarted'; message: ChatMessage }
  | { type: 'sendSucceeded'; clientId: string; message: ChatMessage }
  | { type: 'sendFailed'; clientId: string }
  | { type: 'received'; message: ChatMessage }
  | { type: 'historyLoaded'; messages: ChatMessage[]; hasMore: boolean }
  | { type: 'opened' }
  | { type: 'closed' }
  | { type: 'typing'; active: boolean }
  | { type: 'reset' };

export const initialConversationState: ConversationState = {
  byId: {},
  order: [],
  unread: 0,
  isOpen: false,
  typing: false,
  hasMoreHistory: true,
};

const STATUS_RANK: Record<MessageStatus, number> = { failed: 0, pending: 1, sent: 2, delivered: 3, read: 4 };

function mergeMessage(existing: ChatMessage | undefined, incoming: ChatMessage): ChatMessage {
  if (!existing) return incoming;
  const keepStatus = STATUS_RANK[existing.status] > STATUS_RANK[incoming.status] && incoming.status !== 'failed';
  return {
    ...existing,
    ...incoming,
    clientId: incoming.clientId ?? existing.clientId,
    status: keepStatus ? existing.status : incoming.status,
  };
}

function compareMessages(a: ChatMessage, b: ChatMessage): number {
  const byTime = Date.parse(a.createdAt) - Date.parse(b.createdAt);
  if (byTime !== 0) return byTime;
  return a.id < b.id ? -1 : a.id > b.id ? 1 : 0;
}

/** Inserts or merges messages, deduplicating by server id and by clientId. */
export function upsertMessages(state: ConversationState, messages: ChatMessage[]): { state: ConversationState; added: ChatMessage[] } {
  const byId = { ...state.byId };
  const added: ChatMessage[] = [];

  for (const message of messages) {
    const pendingKey = message.clientId && message.clientId !== message.id && byId[message.clientId] ? message.clientId : undefined;
    const previous = byId[message.id] ?? (pendingKey ? byId[pendingKey] : undefined);
    if (pendingKey) delete byId[pendingKey];
    if (!previous) added.push(message);
    byId[message.id] = mergeMessage(previous, message);
  }

  const order = Object.values(byId).sort(compareMessages).map((m) => m.id);
  return { state: { ...state, byId, order }, added };
}

export function conversationReducer(state: ConversationState, action: ConversationAction): ConversationState {
  switch (action.type) {
    case 'sendStarted':
      return upsertMessages(state, [{ ...action.message, status: 'pending' }]).state;

    case 'sendSucceeded':
      return upsertMessages(state, [{ ...action.message, clientId: action.clientId, status: action.message.status === 'pending' ? 'sent' : action.message.status }]).state;

    case 'sendFailed': {
      const pending = state.byId[action.clientId];
      if (!pending) return state;
      return { ...state, byId: { ...state.byId, [action.clientId]: { ...pending, status: 'failed' } } };
    }

    case 'received': {
      const { state: next, added } = upsertMessages(state, [action.message]);
      const incoming = added.filter((m) => m.sender !== 'user').length;
      return {
        ...next,
        typing: incoming > 0 ? false : next.typing,
        unread: next.isOpen ? 0 : next.unread + incoming,
      };
    }

    case 'historyLoaded':
      return { ...upsertMessages(state, action.messages).state, hasMoreHistory: action.hasMore };

    case 'opened':
      return { ...state, isOpen: true, unread: 0 };

    case 'closed':
      return { ...state, isOpen: false };

    case 'typing':
      return { ...state, typing: action.active };

    case 'reset':
      return initialConversationState;
  }
}

export function selectMessages(state: ConversationState): ChatMessage[] {
  return state.order.map((id) => state.byId[id]).filter((m): m is ChatMessage => m !== undefined);
}
```

Design notes:

- Every insert goes through `upsertMessages`, so history pages, live events, acks and replays after reconnect cannot create duplicates. This closes the reference's gap; see `references/chat-flow.md` A9.
- Status never moves backwards. A `delivered` echo that arrives before the HTTP ack stays `delivered`.
- Order is derived from `createdAt`, not arrival order.

## 5. Store tests (verified)

`tests/conversation.test.ts`

```ts
import { describe, expect, it } from 'vitest';
import { conversationReducer as reduce, initialConversationState as initial, selectMessages } from '../src/state/conversation';
import type { ChatMessage } from '../src/state/types';

const msg = (over: Partial<ChatMessage>): ChatMessage => ({
  id: 'm1', conversationId: 'c1', sender: 'agent', text: 'hi', createdAt: '2026-01-01T10:00:00.000Z', status: 'sent', ...over,
});

describe('conversationReducer', () => {
  it('adds an optimistic message as pending', () => {
    const s = reduce(initial, { type: 'sendStarted', message: msg({ id: 'tmp-1', clientId: 'tmp-1', sender: 'user', status: 'sent' }) });
    expect(selectMessages(s)).toHaveLength(1);
    expect(s.byId['tmp-1']?.status).toBe('pending');
  });

  it('replaces the pending message with the server message on ack', () => {
    let s = reduce(initial, { type: 'sendStarted', message: msg({ id: 'tmp-1', clientId: 'tmp-1', sender: 'user' }) });
    s = reduce(s, { type: 'sendSucceeded', clientId: 'tmp-1', message: msg({ id: 'srv-1', sender: 'user', status: 'sent' }) });
    expect(s.order).toEqual(['srv-1']);
    expect(s.byId['srv-1']).toMatchObject({ clientId: 'tmp-1', status: 'sent' });
  });

  it('does not duplicate when the socket echo arrives before the HTTP ack', () => {
    let s = reduce(initial, { type: 'sendStarted', message: msg({ id: 'tmp-1', clientId: 'tmp-1', sender: 'user' }) });
    s = reduce(s, { type: 'received', message: msg({ id: 'srv-1', sender: 'user', status: 'delivered' }) });
    s = reduce(s, { type: 'sendSucceeded', clientId: 'tmp-1', message: msg({ id: 'srv-1', sender: 'user', status: 'sent' }) });
    expect(s.order).toEqual(['srv-1']);
    expect(s.byId['srv-1']?.status).toBe('delivered'); // never downgraded
  });

  it('matches an echo that carries the clientId', () => {
    let s = reduce(initial, { type: 'sendStarted', message: msg({ id: 'tmp-1', clientId: 'tmp-1', sender: 'user' }) });
    s = reduce(s, { type: 'received', message: msg({ id: 'srv-1', clientId: 'tmp-1', sender: 'user' }) });
    expect(s.order).toEqual(['srv-1']);
  });

  it('marks a failed send and keeps the text', () => {
    let s = reduce(initial, { type: 'sendStarted', message: msg({ id: 'tmp-1', clientId: 'tmp-1', sender: 'user', text: 'keep me' }) });
    s = reduce(s, { type: 'sendFailed', clientId: 'tmp-1' });
    expect(s.byId['tmp-1']).toMatchObject({ status: 'failed', text: 'keep me' });
  });

  it('retry after failure goes back to pending', () => {
    let s = reduce(initial, { type: 'sendStarted', message: msg({ id: 'tmp-1', clientId: 'tmp-1', sender: 'user' }) });
    s = reduce(s, { type: 'sendFailed', clientId: 'tmp-1' });
    s = reduce(s, { type: 'sendStarted', message: s.byId['tmp-1']! });
    expect(s.byId['tmp-1']?.status).toBe('pending');
  });

  it('orders by createdAt regardless of arrival order', () => {
    let s = reduce(initial, { type: 'received', message: msg({ id: 'b', createdAt: '2026-01-01T10:00:02.000Z' }) });
    s = reduce(s, { type: 'received', message: msg({ id: 'a', createdAt: '2026-01-01T10:00:01.000Z' }) });
    expect(s.order).toEqual(['a', 'b']);
  });

  it('merges history pages without duplicating live messages', () => {
    let s = reduce(initial, { type: 'received', message: msg({ id: 'm3', createdAt: '2026-01-01T10:00:03.000Z' }) });
    s = reduce(s, { type: 'historyLoaded', hasMore: false, messages: [
      msg({ id: 'm1', createdAt: '2026-01-01T10:00:01.000Z' }),
      msg({ id: 'm2', createdAt: '2026-01-01T10:00:02.000Z' }),
      msg({ id: 'm3', createdAt: '2026-01-01T10:00:03.000Z' }),
    ] });
    expect(s.order).toEqual(['m1', 'm2', 'm3']);
    expect(s.hasMoreHistory).toBe(false);
  });

  it('counts unread only for new incoming messages while closed', () => {
    let s = reduce(initial, { type: 'received', message: msg({ id: 'a' }) });
    s = reduce(s, { type: 'received', message: msg({ id: 'a' }) }); // replayed after reconnect
    s = reduce(s, { type: 'received', message: msg({ id: 'u', sender: 'user' }) });
    expect(s.unread).toBe(1);
    s = reduce(s, { type: 'opened' });
    expect(s.unread).toBe(0);
    s = reduce(s, { type: 'received', message: msg({ id: 'b' }) });
    expect(s.unread).toBe(0);
  });

  it('clears the typing indicator when a reply arrives', () => {
    let s = reduce(initial, { type: 'typing', active: true });
    s = reduce(s, { type: 'received', message: msg({ id: 'r1', sender: 'bot' }) });
    expect(s.typing).toBe(false);
  });
});
```

## 6. Wiring outline: controller, Web Component, loader

The controller is the only place that talks to the API, the transport and the store:

```ts
// src/chat-controller.ts (outline)
const api = createChatApi(createApiClient({ baseUrl, getAuthHeaders: () => ({ Authorization: `Bearer ${session.token}` }) }));
let state = initialConversationState;
const dispatch = (action: ConversationAction) => { state = conversationReducer(state, action); render(state); emitUnread(state.unread); };

async function start(user?: IdentifiedUser) {
  const s = await api.startSession(config.widgetId, user);           // anonymous when user is undefined
  saveSession(s);                                                     // namespaced, safe storage
  const page = await api.fetchMessages(s.conversationId);
  dispatch({ type: 'historyLoaded', ...page });
  realtime = connectRealtime(config.socketUrl, s.sessionToken, s.conversationId, {
    onMessage: (m) => dispatch({ type: 'received', message: m }),
    onTyping: (active) => dispatch({ type: 'typing', active }),
    onConnectionChange: (up) => setBanner(up ? null : 'Reconnecting…'),
    onReconnected: () => api.fetchMessages(s.conversationId).then((p) => dispatch({ type: 'historyLoaded', ...p })),
  });
}

async function send(text: string) {
  const clientId = crypto.randomUUID();
  dispatch({ type: 'sendStarted', message: { id: clientId, clientId, conversationId, sender: 'user', text, createdAt: new Date().toISOString(), status: 'pending' } });
  try { dispatch({ type: 'sendSucceeded', clientId, message: await api.sendMessage(conversationId, clientId, text) }); }
  catch { dispatch({ type: 'sendFailed', clientId }); }
}
```

UI (`src/ui/acme-chat.ts`): a custom element with an open shadow root, where `render(state)` patches only what changed:

- **Launcher:** a `<button aria-expanded>` with an unread badge.
- **Window:** `role="dialog"`.
- **Message list:** `role="log"`, `aria-live="polite"`, with an IntersectionObserver at the top that loads older history.
- **Bubbles:** text goes through `textContent`. Markdown goes through `DOMPurify.sanitize(marked.parse(text))` with links rewritten to `rel="noopener noreferrer" target="_blank"`.
- **States:** typing dots, a failed bubble with a Retry button, and the empty state showing the welcome message.
- **Composer:** Enter sends and Shift+Enter inserts a newline.

Loader (`src/index.ts`): `window.AcmeChat = { init(config) }`. `init` validates the config, creates `<acme-chat>`, appends it to `document.body` and returns `{ open, close, identify, logout, destroy }`.
